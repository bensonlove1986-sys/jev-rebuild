## 🎯 执行目标

实现一个自动报价引擎（AutoQuote Engine），  
让系统在收到客户图片或文本（通过 Daojie Webhook）时，能自动：


1. 识别型号 / 品牌 / 数量  
2. 从 Excel 价格表中查询匹配价格  
3. 计算报价（含利润率、包装费等）  
4. 生成 JSON 或表格格式的报价结果  
5. 通过 Daojie API 自动回复客户

## ⚙️ 操作步骤

1️⃣ 新增 `/auto-quote` 接口；  
2️⃣ 从请求体中读取：`image_url` 或 `message_text`；  
3️⃣ 调用 Local AI 服务识别型号：

```


POST http://localhost:1822/analyze


````
响应示例：

```json

{ "model": "E2E-X3D1-N", "brand": "OMRON", "qty": 12 }

```


4️⃣ 从 Excel 价格表中匹配：


* 价格表存于 KV 或 R2，例如：`price_db.xlsx`；
* 字段：Model | Brand | BasePrice | Currency；
 5️⃣ 计算最终报价：


```

FinalPrice = BasePrice * (1 + MarginRate) + Packaging + Shipping

```


例如：


```

3.6 * (1 + 0.25) + 0.3 = 4.8 USD

```


6️⃣ 生成输出：


```json

{

  "model": "E2E-X3D1-N",

  "brand": "OMRON",

  "qty": 12,

  "unit_price": 4.8,

  "total": 57.6,

  "currency": "USD"

}

```


7️⃣ 通过 Daojie Webhook 发送：


```json

{

  "event": "auto_quote",

  "contact_id": "...",

  "message": "E2E-X3D1-N (OMRON)\nUnit: $4.8\nTotal: $57.6\nAuto-generated by JEV"

}

```


8️⃣ 保存结果到：


* `quote:<date>:<contact_id>`；
* `log:quote:<date>`。

## 💻 Worker 代码模板


```js

// === JEV AutoQuote Engine ===

if (path === "/auto-quote") {

  const result = await runAutoQuote(request, env);

  return new Response(JSON.stringify(result, null, 2), {

    headers: { "Content-Type": "application/json" },

    status: 200,

  });

}


async function runAutoQuote(request, env) {

  const body = await request.json();

  const { image_url, message_text, contact_id } = body;


  // Step 1: AI识别

  let analyze = {};

  if (image_url) {

    const aiRes = await fetch("http://localhost:1822/analyze", {

      method: "POST",

      headers: { "Content-Type": "application/json" },

      body: JSON.stringify({ image_url }),

    });

    analyze = await aiRes.json();

  } else if (message_text) {

    const aiRes = await fetch("http://localhost:1822/extract-text", {

      method: "POST",

      headers: { "Content-Type": "application/json" },

      body: JSON.stringify({ text: message_text }),

    });

    analyze = await aiRes.json();

  }


  const { model, brand, qty } = analyze;

  if (!model) return { success: false, error: "No model recognized" };


  // Step 2: 查询价格表

  const priceDb = await env.JEVPI.get("price_db", { type: "json" });

  const item = priceDb?.find(

    (p) => p.Model === model || p.Model.includes(model)

  );


  if (!item) return { success: false, error: "Model not found in DB" };


  // Step 3: 计算报价

  const margin = parseFloat(env.MARGIN_RATE || "0.25");

  const packaging = parseFloat(env.PACKAGING_FEE || "0.3");

  const unitPrice = item.BasePrice * (1 + margin) + packaging;

  const total = unitPrice * (qty || 1);


  const quote = {

    model,

    brand: brand || item.Brand,

    qty: qty || 1,

    unit_price: unitPrice.toFixed(2),

    total: total.toFixed(2),

    currency: item.Currency || "USD",

  };


  // Step 4: 存储日志

  const today = new Date().toISOString().split("T")[0];

  const key = `log:quote:${today}`;

  const prev = await env.JEVPI.get(key, { type: "json" }) || [];

  prev.push({ time: new Date().toISOString(), ...quote });

  await env.JEVPI.put(key, JSON.stringify(prev));


  // Step 5: 回复客户（Daojie）

  if (env.DAOJIE_WEBHOOK_URL && contact_id) {

    await fetch(env.DAOJIE_WEBHOOK_URL, {

      method: "POST",

      headers: { "Content-Type": "application/json" },

      body: JSON.stringify({

        event: "auto_quote",

        contact_id,

        message: `${model} (${quote.brand})\nUnit: $${quote.unit_price}\nTotal: $${quote.total}`,

      }),

    });

  }


  return { success: true, quote };

}

```
