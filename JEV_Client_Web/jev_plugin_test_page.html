基本用法

// 初始化插件 const jevPlugin = new JEVPluginSync({ workerUrl: 'https://jevshop.bensonlove1986.workers.dev', agentToken: 'jev_cloud_api_key_123456' }); // 运行对齐测试 await jevPlugin.runAlignmentTest(); // 发送消息 const response = await jevPlugin.sendMessage({ device: 'plugin-ui', key: 'page_event', value: { data: '自定义数据' }, timestamp: Date.now(), origin: 'plugin' }); // 获取状态 const status = jevPlugin.getModeStatus();
