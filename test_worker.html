正确的 Worker URL 配置

// 在 Tampermonkey 脚本中设置正确的 Worker URL
window.JEV.STATE.workerUrl = 'https://jevshop.bensonlove1986.workers.dev';

// 认证配置 (Bearer Token 方式)
window.JEV.STATE.authToken = 'jevindustry2025';

// 在请求中使用认证
headers: {
    'Authorization': `Bearer ${window.JEV.STATE.authToken}`
}

连接检查步骤

  1. 确保 Worker URL 正确无误
  2. 验证认证密钥是 jevindustry2025
  3. 使用上方工具测试 /ping 端点，应返回 {"success":true,"pong":true,...}
  4. 在 Tampermonkey 脚本中设置相同的配置
  5. 清除浏览器缓存后重新加载页面

常见问题排查

  • 如果 /ping 端点返回 401，检查认证密钥是否正确
  • 如果返回 500 错误，检查 Worker 日志中的错误信息
  • 如果连接超时，可能是网络问题或 Worker 部署失败
  • 确保使用 HTTPS 而不是 HTTP
  • 检查浏览器的跨域设置和安全策略

为什么会有 CORS 问题？

Tampermonkey 脚本运行在浏览器沙盒环境中，默认受到 CORS（跨域资源共享）限制。即使 Worker 设置了 Access-Control-Allow-Origin: *，在某些情况下（特别是开发域名）仍然可能失败。

方案 1: GM_xmlhttpRequest

使用 Tampermonkey 的专用 API，完全绕过浏览器 CORS 限制

最推荐的解决方案，无需修改 Worker 配置
方案 2: 使用正式域名

使用正式 Worker 域名而非开发域名，减少 CORS 问题

推荐使用: jevshop.bensonlove1986.workers.dev
方案 3: 配置 @connect

确保在脚本元数据中正确配置域名连接权限

添加: @connect jevshop.bensonlove1986.workers.dev

Worker 已优化的 CORS 配置

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS, PUT, DELETE, PATCH',
  'Access-Control-Allow-Headers': '*', // 允许所有头信息
  'Access-Control-Expose-Headers': '*', // 暴露所有响应头
  'Access-Control-Allow-Credentials': 'true', // 支持凭证
  'Access-Control-Max-Age': '86400'
};

GM_xmlhttpRequest 基础模板 

// ==UserScript==
// @name         Cloudflare Worker 访问脚本
// @namespace    http://tampermonkey.net/
// @version      0.1
// @description  使用GM_xmlhttpRequest访问Cloudflare Worker，绕过CORS限制
// @author       You
// @match        https://app.daojiemarketing.com/*
// @grant        GM_xmlhttpRequest
// @connect      jevshop.bensonlove1986.workers.dev
// ==/UserScript==

(function() {
    'use strict';
    
    // Worker配置
    const WORKER_URL = 'https://jevshop.bensonlove1986.workers.dev';
    const API_KEY = 'jevindustry2025';
    
    // 测试Worker连接
    function testWorkerConnection() {
        console.log('测试Worker连接...');
        
        GM_xmlhttpRequest({
            method: "GET",
            url: WORKER_URL + "/ping",
            headers: {
                "Content-Type": "application/json",
                "X-Webhook-Secret": API_KEY
            },
            onload: function(response) {
                console.log("✅ Worker连接成功!");
                try {
                    const data = JSON.parse(response.responseText);
                    console.log("响应数据:", data);
                } catch (e) {
                    console.log("响应文本:", response.responseText);
                }
            },
            onerror: function(error) {
                console.error("❌ Worker连接失败:", error);
            },
            ontimeout: function() {
                console.error("❌ 请求超时");
            }
        });
    }
    
    // 分析图像示例
    function analyzeImage(imageData) {
        GM_xmlhttpRequest({
            method: "POST",
            url: WORKER_URL + "/analyze-image",
            headers: {
                "Content-Type": "application/json",
                "X-Webhook-Secret": API_KEY
            },
            data: JSON.stringify({
                image_id: imageData.id,
                analysis_type: "basic"
                // 可以添加更多参数
            }),
            onload: function(response) {
                console.log("图像分析结果:", JSON.parse(response.responseText));
            },
            onerror: function(error) {
                console.error("图像分析失败:", error);
            }
        });
    }
    
    // 当页面加载完成时执行
    window.addEventListener('load', function() {
        console.log('Worker API客户端已初始化');
        // 测试连接
        testWorkerConnection();
        
        // 在这里添加您的业务逻辑
    });
})();

Promise 封装版本 (推荐) 

// ==UserScript==
// @name         Cloudflare Worker Promise API
// @namespace    http://tampermonkey.net/
// @version      0.1
// @description  使用Promise封装GM_xmlhttpRequest，提供现代异步接口
// @author       You
// @match        https://app.daojiemarketing.com/*
// @grant        GM_xmlhttpRequest
// @connect      jevshop.bensonlove1986.workers.dev
// ==/UserScript==

(function() {
    'use strict';
    
    // Worker配置
    const WORKER_URL = 'https://jevshop.bensonlove1986.workers.dev';
    const API_KEY = 'jevindustry2025';
    
    // Promise封装GM_xmlhttpRequest
    function callWorker(endpoint, method = 'GET', data = null) {
        return new Promise((resolve, reject) => {
            const options = {
                method: method,
                url: `${WORKER_URL}${endpoint}`,
                headers: {
                    'Content-Type': 'application/json',
                    'X-Webhook-Secret': API_KEY
                },
                onload: function(response) {
                    try {
                        const result = JSON.parse(response.responseText);
                        resolve(result);
                    } catch (e) {
                        reject(new Error('无效的JSON响应'));
                    }
                },
                onerror: function(error) {
                    reject(new Error(`请求失败: ${error.message}`));
                },
                ontimeout: function() {
                    reject(new Error('请求超时'));
                }
            };
            
            if (data && method !== 'GET') {
                options.data = JSON.stringify(data);
            }
            
            GM_xmlhttpRequest(options);
        });
    }
    
    // 异步测试函数
    async function testWorkerAsync() {
        try {
            console.log('开始异步测试...');
            
            // 测试ping端点
            const pingResult = await callWorker('/ping');
            console.log('✅ Ping成功:', pingResult);
            
            // 测试分析图像
            const analysisResult = await callWorker('/analyze-image', 'POST', {
                image_id: 'test_image_123',
                analysis_type: 'basic'
            });
            console.log('✅ 图像分析成功:', analysisResult);
            
            return { ping: pingResult, analysis: analysisResult };
        } catch (error) {
            console.error('❌ 测试失败:', error.message);
            throw error;
        }
    }
    
    // 使用示例
    window.addEventListener('load', async function() {
        console.log('Worker Promise API已初始化');
        
        try {
            await testWorkerAsync();
            console.log('所有测试完成!');
        } catch (error) {
            console.error('测试过程中出错:', error);
        }
    });
    
    // 暴露API给全局作用域
    window.WorkerAPI = {
        call: callWorker,
        testConnection: () => callWorker('/ping'),
        analyzeImage: (data) => callWorker('/analyze-image', 'POST', data)
    };
})();
